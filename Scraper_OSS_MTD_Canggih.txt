    async function startCrawling() {
        isRunning = true;

        let targetTable = null;

                window.location.href = BOOKMARK_URLS[currentBmkIdx];
            }, delayMs);
            return;
        }

        let headerCells = targetTable.querySelectorAll('tr')[0].querySelectorAll('th, td');
        // Bersihkan header dari enter (\n) yang bikin gagal deteksi
        headers = Array.from(headerCells).map(c => c.innerText.trim().replace(/\s+/g, ' '));

            // CLOSED TICKET: Prioritaskan STATUS DATE
            reportedDateIdx = headers.findIndex(h => h.includes('STATUS DATE') || h.includes('STATUSDATE'));
            if(reportedDateIdx === -1) {
                reportedDateIdx = headers.findIndex(h => h.includes('REPORTED DATE'));
            }
        } else {
            // ALL TICKET: Prioritaskan STATUS DATE agar sinkron dengan sorting user
            reportedDateIdx = headers.findIndex(h => h.includes('STATUS DATE') || h.includes('STATUSDATE'));
            if(reportedDateIdx === -1) {
                reportedDateIdx = headers.findIndex(h => h.includes('REPORTED DATE') || h.includes('REPORTEDDATE'));
            }

        // Jika masih tidak ada, cari yang ada kata DATE

        if(reportedDateIdx === -1) {

            let delayStr = (currentBmkIdx === 1) ? "10 detik" : "10 menit";

            isRunning = false;
            
            setTimeout(() => {

                if(currentBmkIdx >= BOOKMARK_URLS.length) currentBmkIdx = 0;
                localStorage.setItem('oss_scraper_bmk_idx', currentBmkIdx);
                window.location.href = BOOKMARK_URLS[currentBmkIdx];
            }, delayMs);
            return;
        }

        scrapedData.push(headers); 
        let hasOlderOnFirstPage = extractAndFilterRows(targetTable);
        if(hasOlderOnFirstPage) {
            sendToGoogleSheet();
            return;
        }
        
        let pagelinks = document.querySelector('.pagelinks a') || document.querySelector('a[href*="d-"]');
        let paramPrefix = "";
        
        if (pagelinks) {
            let href = pagelinks.getAttribute('href');
            let match = href.match(/(d-\d+-)/);
            if (match) paramPrefix = match[1];
        }
        
        // Fallback: cari dari URL saat ini (contoh: url?d-5564009-p=1)
        if (!paramPrefix) {
            let urlMatch = window.location.href.match(/(d-\d+-)/);
            if (urlMatch) {
                paramPrefix = urlMatch[1];
            } else {
                sendToGoogleSheet();
                return;
            }
        }

        let baseUrl = window.location.origin + window.location.pathname; 
        let currentPageNum = 2; 
        let totalHalaman = 264; // Default aman jika gagal baca info

        // Baca jumlah info (Contoh: "26,220 items found")
        try {
            let infoDiv = document.querySelector('.pagelinks') ? document.querySelector('.pagelinks').parentNode.innerText : document.body.innerText;
            let totalItemsMatch = infoDiv.match(/([\d,]+)\s+items found/);
            if(totalItemsMatch) {
                let total = parseInt(totalItemsMatch[1].replace(/,/g, ''));
                totalHalaman = Math.ceil(total / 100);
            }
        } catch(e) {
            console.log("Gagal membaca total halaman, menggunakan default", e);
        }

        while (currentPageNum <= totalHalaman) {
            setStatus(`⏳ Mengekstrak Halaman <b>${currentPageNum}</b> dari <b>${totalHalaman}</b>...<br>Tiket Hari Ini ditemukan: <b>${scrapedData.length - 1}</b>`);
            
            let targetUrl = new URL(baseUrl);
            targetUrl.searchParams.set(paramPrefix + 'p', currentPageNum);
            targetUrl.searchParams.set(paramPrefix + 'ps', '100'); 

            try {
                let response = await fetch(targetUrl.toString());
                let htmlText = await response.text();
                
                let doc = new DOMParser().parseFromString(htmlText, 'text/html');
                let shadowTables = doc.querySelectorAll('table');
                let shadowTarget = null;
                let shadowMax = 0;
                
                shadowTables.forEach(t => {
                    let trs = t.querySelectorAll('tr');
                    if(trs.length > shadowMax) { shadowMax = trs.length; shadowTarget = t; }
                });

                if (!shadowTarget || shadowMax < 2) {
                    break; 
                }

                let hasOlder = extractAndFilterRows(shadowTarget);
                
                if (hasOlder) {
                    setStatus(`⚠️ Menemukan tiket dari hari sebelumnya. Menghentikan pencarian halaman.`);
                    break;
                }

                currentPageNum++;
                await new Promise(r => setTimeout(r, 400)); 
            } catch (err) {
                setStatus(`❌ Error di hal ${currentPageNum}. Melompat ke pengiriman...`);
                break;
            }
        }

        sendToGoogleSheet();
    }

    function extractAndFilterRows(tableElem) {
        let rows = tableElem.querySelectorAll('tr');
        let foundOlderDate = false;
        
        for (let i = 1; i < rows.length; i++) {
            let cells = rows[i].querySelectorAll('td');
            if(cells.length > 0) {
                let rowData = Array.from(cells).map(c => c.innerText.trim());
                
                // Jika tabel benar-benar kosong, hentikan pencarian halaman berikutnya
                if(rowData.length === 1 && (rowData[0].includes('Nothing found') || rowData[0].includes('No items found'))) {
                    return true; 
                }

                let dateStr = "";
                if (reportedDateIdx > -1 && rowData[reportedDateIdx]) {
                    dateStr = rowData[reportedDateIdx].replace(/\s+/g, '');
                }
                
                // Jika kolom header kosong (sering terjadi), cari cell manapun yang mengandung format tanggal YYYY-MM-DD
                if (!dateStr || !dateStr.match(/\d{4}-\d{2}-\d{2}/)) {
                    let fallbackCell = rowData.find(c => c.match(/\d{4}-\d{2}-\d{2}/));
                    if (fallbackCell) {
                        dateStr = fallbackCell.replace(/\s+/g, '');
                    }
                }
                
                let targetStr = CURRENT_DATE.replace(/\s+/g, ''); // "2026-09-27"
                
                // AMBIL SEMUA ROW YANG ADA DI TABEL!
                // Jangan terlalu ketat pada filter tanggal karena OSS sering return tiket kemarin sore karena beda zona waktu!
                scrapedData.push(rowData);
                
                if (dateStr) {
                    // Cek apakah tanggal di baris ini bukan hari ini, jika ya, hentikan pencarian halaman berikutnya
                    let rowDateMatch = dateStr.match(/(\d{4}-\d{2}-\d{2})/);
                    if (rowDateMatch) {
                        let rowDateStr = rowDateMatch[1];
                        if (rowDateStr !== targetStr) {
                            foundOlderDate = true;
                        }
                    }
                }
            }
        }
        return foundOlderDate;
    }

            data: scrapedData

            method: 'POST',

                if(waitTime < 0) {
                    clearInterval(timerInterval);
                    currentBmkIdx++;
                    if(currentBmkIdx >= BOOKMARK_URLS.length) currentBmkIdx = 0;
                    localStorage.setItem('oss_scraper_bmk_idx', currentBmkIdx);
                    window.location.href = BOOKMARK_URLS[currentBmkIdx];
                }
            }, 1000);
        }).catch(err => {
            let delayMs = (currentBmkIdx === 1) ? 10000 : 600000;
            let delayStr = (currentBmkIdx === 1) ? "10 detik" : "10 menit";
            setStatus(`❌ Gagal mengirim data ke Google Sheet.<br>Lanjut ke bookmark berikutnya dalam ${delayStr}...`);
            setTimeout(() => {
                currentBmkIdx++;
                if(currentBmkIdx >= BOOKMARK_URLS.length) currentBmkIdx = 0;
                localStorage.setItem('oss_scraper_bmk_idx', currentBmkIdx);
                window.location.href = BOOKMARK_URLS[currentBmkIdx];
            }, delayMs);
        });
    }

    function runMain() {
        // Jika sedang di halaman utama atau sembarang halaman setelah login
        if (!isTargetPage) {
            let container = document.createElement('div');
            container.style.cssText = `position: fixed; top: 20px; right: 20px; background: #0284c7; color: white; padding: 15px; border-radius: 8px; z-index: 999999; box-shadow: 0 4px 15px rgba(0,0,0,0.2); font-family: Arial; text-align: center;`;
            container.innerHTML = `🚀 <b>GENCU SCRAPER AKTIF</b><br><small>Sistem akan melompat ke bookmark target dalam 5 detik...</small>`;
            document.body.appendChild(container);
            
            setTimeout(() => {
                window.location.href = BOOKMARK_URLS[currentBmkIdx];
            }, 5000);
            return;
        }

        setTimeout(() => {
            initUI();
            
            let isEmpty = false;
            let pageText = document.body.innerText;
            if (pageText.includes('Nothing found') || pageText.includes('No items found') || pageText.includes('Please Submit Filters')) {
                isEmpty = true;
            }

            // AUTO-FILTER KHUSUS HALAMAN CLOSED TICKET
            if (window.location.href.includes('allTicketListRepo') && isEmpty && !sessionStorage.getItem('oss_auto_filtered')) {
                setStatus("Mencoba filter otomatis untuk Hari Ini...");
                sessionStorage.setItem('oss_auto_filtered', 'true');

                let filterApplied = false;
                
                // Cari input tanggal berdasarkan icon kalender di sebelahnya
                let dateInputs = Array.from(document.querySelectorAll('input[type="text"]')).filter(input => {
                    let nextEl = input.nextElementSibling;
                    return (nextEl && nextEl.tagName === 'IMG' && nextEl.src.toLowerCase().includes('calendar'));
                });
                
                // Jika ketemu 4 (Reported From-To, Status From-To), isi yang Status saja
                if (dateInputs.length >= 4) {
                    dateInputs[2].value = CURRENT_DATE;
                    dateInputs[3].value = CURRENT_DATE;
                    filterApplied = true;
                } else if (dateInputs.length > 0) {
                    // Fallback, isi semuanya
                    dateInputs.forEach(i => i.value = CURRENT_DATE);
                    filterApplied = true;
                } else {
                    // Fallback super darurat
                    let allDateInputs = Array.from(document.querySelectorAll('input[type="text"]')).filter(i => 
                        (i.placeholder && (i.placeholder.includes('From') || i.placeholder.includes('To')))
                    );
                    if (allDateInputs.length >= 4) {
                        allDateInputs[2].value = CURRENT_DATE;
                        allDateInputs[3].value = CURRENT_DATE;
                        filterApplied = true;
                    }
                }

                if (filterApplied) {
                    let showBtn = Array.from(document.querySelectorAll('input, button')).find(el => 
                        (el.value && el.value.toLowerCase() === 'show') || 
                        (el.innerText && el.innerText.toLowerCase().trim() === 'show')
                    );

                    if (showBtn) {
                        setStatus("Filter diisi. Mengklik 'show'...");
                        setTimeout(() => showBtn.click(), 1000);
                        return; // Stop eksekusi, biarkan halaman reload
                    }
                }
            }

            // Jika kita ada di ALL TICKET, atau tabel tidak kosong, reset flag
            if (!isEmpty || window.location.href.includes('allTicketList')) {
                if (!window.location.href.includes('allTicketListRepo')) {
                    sessionStorage.removeItem('oss_auto_filtered');
                }
            }

            // FITUR AUTO START (Akan jalan meskipun tabel kosong, supaya dia bisa melompat ke bookmark berikutnya)
            let btn = document.querySelector('#oss-canggih-panel button');
            if (btn && !isRunning) {
                setStatus("Sistem siap. Otomatis memulai dalam 5 detik...");
                setTimeout(() => {
                    btn.click();
                }, 5000);
            }
        }, 1500);
    }

    if (document.readyState === 'complete' || document.readyState === 'interactive') {
        runMain();
    } else {
        window.addEventListener('DOMContentLoaded', runMain);
    }
})();
